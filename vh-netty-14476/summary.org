** netty issue draft
#+begin_src markdown
FAO @jonas

Netty can't enable VarHandle in native image due to performance issues.
One of those issues is https://github.com/oracle/graal/issues/14476

If `PlatformDependant.VAR_HANDLE` could be initialized at build time,
the issue above could be worked around.

To resolve `VAR_HANDLE` Netty checks the java version (which also checks if it's Android),
checks whether the platform supports alignment,
and checks if VarHandle can be correctly instantiated.

It's unclear if all of these checks could be deferred to build time.
#+end_src
** graal issue draft
#+begin_src markdown
Netty's migration from Unsafe to VarHandle has not activated in native image due to performance issues.
Under certain conditions, native image ends up using `MethodHandle::linkToStatic` to execute VarHandle instead of intrisifying the call.
This results is really poor performance.

Netty has a static final boolean variable to see if VarHandle is available or not.
Even though it's final, the GraalVM compiler is not treating that as a constant when the class holding it is runtime init.

# Background

https://github.com/Songdoeon/netty-varhandle-bench
which you can run with my JMH extension for GraalVM native image called [Fibula](https://github.com/galderz/fibula).

**NOTE:** `netty.use-new-code` system property activates VarHandle even if its disables by default in Netty.

The bulk of the time is being spent in Netty's `VarHandleByteBufferAccess.getLongLE` which ends up calling `MethodHandle::linkToStatic`.

After some experimentation I ended up replicating that scenario with the following vanilla Java JMH benchmark (see [here](https://github.com/galderz/fibula-show/tree/main/2609-handles) for full project):

```java
package org.sample.handles;

import org.openjdk.jmh.annotations.Benchmark;
import org.openjdk.jmh.annotations.BenchmarkMode;
import org.openjdk.jmh.annotations.Fork;
import org.openjdk.jmh.annotations.Measurement;
import org.openjdk.jmh.annotations.Mode;
import org.openjdk.jmh.annotations.OutputTimeUnit;
import org.openjdk.jmh.annotations.Param;
import org.openjdk.jmh.annotations.Scope;
import org.openjdk.jmh.annotations.Setup;
import org.openjdk.jmh.annotations.State;
import org.openjdk.jmh.annotations.Warmup;

import java.lang.invoke.MethodHandle;
import java.lang.invoke.MethodHandles;
import java.lang.invoke.MethodType;
import java.lang.invoke.VarHandle;
import java.nio.ByteOrder;
import java.util.Random;
import java.util.concurrent.TimeUnit;

@BenchmarkMode(Mode.Throughput)
@OutputTimeUnit(TimeUnit.MILLISECONDS)
@State(Scope.Thread)
@Warmup(iterations = 10, time = 500, timeUnit = TimeUnit.MILLISECONDS)
@Measurement(iterations = 5, time = 1000, timeUnit = TimeUnit.MILLISECONDS)
@Fork(1)
public class NettyByteArrays
{
    @Param({"256"})
    private int size;

    byte[] b;

    int index;

    @Setup
    public void setup()
    {
        final Random r = new Random(42);

        b = new byte[size];
        for (int i = 0; i < b.length; i++)
        {
            b[i] = (byte) r.nextInt();
        }

        index = 0;
    }

    @Benchmark
    public long plainGetLongLE()
    {
        return ((long) b[index] & 0xFF) |
            ((long) b[index + 1] & 0xFF) << 8 |
            ((long) b[index + 2] & 0xFF) << 16 |
            ((long) b[index + 3] & 0xFF) << 24 |
            ((long) b[index + 4] & 0xFF) << 32 |
            ((long) b[index + 5] & 0xFF) << 40 |
            ((long) b[index + 6] & 0xFF) << 48 |
            ((long) b[index + 7] & 0xFF) << 56;
    }

    private static final boolean VAR_HANDLE;
    private static final VarHandle LONG_LE_ARRAY_VIEW;

    static
    {
        VAR_HANDLE = initializeVarHandle();

        VarHandle longLeArrayViewHandle = null;
        try
        {
            MethodHandles.Lookup lookup = MethodHandles.lookup();
            MethodHandle byteArrayViewHandle = lookup.findStatic(
                MethodHandles.class
                , "byteArrayViewVarHandle"
                , MethodType.methodType(VarHandle.class, Class.class, ByteOrder.class)
            );

            longLeArrayViewHandle = (VarHandle) byteArrayViewHandle.invokeExact(long[].class, ByteOrder.LITTLE_ENDIAN);
        }
        catch (Throwable e)
        {
            longLeArrayViewHandle = null;
        }
        finally
        {
            LONG_LE_ARRAY_VIEW = longLeArrayViewHandle;
        }
    }

    @Benchmark
    public long vhandleGetLongLE()
    {
        VarHandle vh = null;
        if (VAR_HANDLE) {
            vh = LONG_LE_ARRAY_VIEW;
        }

        return (long) vh.get(b, index);
    }

    private static boolean initializeVarHandle()
    {
        if (javaVersion0() < 9) {
            return false;
        }

        return true;
    }

    private static int javaVersion0()
    {
        final int majorVersion;
        majorVersion = majorVersionFromJavaSpecificationVersion();
        System.out.println("Java version: " + majorVersion);
        return majorVersion;
    }

    private static int majorVersionFromJavaSpecificationVersion()
    {
        return majorVersion(SystemPropertyUtil.get("java.specification.version", "1.6"));
    }

    private static int majorVersion(final String javaSpecVersion)
    {
        final String[] components = javaSpecVersion.split("\\.");
        final int[] version = new int[components.length];
        for (int i = 0; i < components.length; i++)
        {
            version[i] = Integer.parseInt(components[i]);
        }

        if (version[0] == 1)
        {
            assert version[1] >= 6;
            return version[1];
        }
        else
        {
            return version[0];
        }
    }
}
```

How good could this benchmark be in the VarHandle was intrinsified?
To answer this question, we can use this separate benchmark where the VarHandle usage is not conditional:

```java
package org.sample.handles;

import org.openjdk.jmh.annotations.Benchmark;
import org.openjdk.jmh.annotations.BenchmarkMode;
import org.openjdk.jmh.annotations.Fork;
import org.openjdk.jmh.annotations.Measurement;
import org.openjdk.jmh.annotations.Mode;
import org.openjdk.jmh.annotations.OutputTimeUnit;
import org.openjdk.jmh.annotations.Param;
import org.openjdk.jmh.annotations.Scope;
import org.openjdk.jmh.annotations.Setup;
import org.openjdk.jmh.annotations.State;
import org.openjdk.jmh.annotations.Warmup;

import java.lang.invoke.MethodHandle;
import java.lang.invoke.MethodHandles;
import java.lang.invoke.MethodType;
import java.lang.invoke.VarHandle;
import java.nio.ByteOrder;
import java.util.Random;
import java.util.concurrent.TimeUnit;

@BenchmarkMode(Mode.Throughput)
@OutputTimeUnit(TimeUnit.MILLISECONDS)
@State(Scope.Thread)
@Warmup(iterations = 10, time = 500, timeUnit = TimeUnit.MILLISECONDS)
@Measurement(iterations = 5, time = 1000, timeUnit = TimeUnit.MILLISECONDS)
@Fork(1)
public class VanillaByteArrays
{
    private static final VarHandle LONG_LE_ARRAY_VIEW;

    @Param({"256"})
    private int size;

    byte[] b;

    int index;

    static
    {
        VarHandle longLeArrayViewHandle = null;
        try
        {
            MethodHandles.Lookup lookup = MethodHandles.lookup();
            MethodHandle byteArrayViewHandle = lookup.findStatic(
                MethodHandles.class
                , "byteArrayViewVarHandle"
                , MethodType.methodType(VarHandle.class, Class.class, ByteOrder.class)
            );

            longLeArrayViewHandle = (VarHandle) byteArrayViewHandle.invokeExact(long[].class, ByteOrder.LITTLE_ENDIAN);
        }
        catch (Throwable e)
        {
            longLeArrayViewHandle = null;
        }
        finally
        {
            LONG_LE_ARRAY_VIEW = longLeArrayViewHandle;
        }
    }

    @Setup
    public void setup()
    {
        final Random r = new Random(42);

        b = new byte[size];
        for (int i = 0; i < b.length; i++)
        {
            b[i] = (byte) r.nextInt();
        }

        index = 0;
    }

    @Benchmark
    public long vhandleGetLongLE()
    {
        return (long) LONG_LE_ARRAY_VIEW.get(b, index);
    }

    @Benchmark
    public long plainGetLongLE()
    {
        return ((long) b[index] & 0xFF) |
            ((long) b[index + 1] & 0xFF) << 8 |
            ((long) b[index + 2] & 0xFF) << 16 |
            ((long) b[index + 3] & 0xFF) << 24 |
            ((long) b[index + 4] & 0xFF) << 32 |
            ((long) b[index + 5] & 0xFF) << 40 |
            ((long) b[index + 6] & 0xFF) << 48 |
            ((long) b[index + 7] & 0xFF) << 56;
    }
}
```
